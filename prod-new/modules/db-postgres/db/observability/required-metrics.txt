# required-metrics.txt - the contract: every name here MUST be present in a scrape of
# the exporter against a freshly bootstrapped database (scripts/db-scrape-test.sh
# fails on an absent name). Format: `name   # why`. A name no alert uses MUST say
# `dashboard-only` in its comment (parity-check.sh fails an orphan); a name an alert
# uses is required, or the alert could never fire and nobody would know.
#
# Names are the exporter's v0.21.1 names as documented; the ones marked UNVERIFIED
# were not seen in a real scrape when this file was written. The first scrape test
# in containers settles them: fix the name here AND in the alert, never drop it.
pg_up                                                 # availability; alert PostgresDown; absent() of it drives PostgresExporterAbsent
pg_scrape_collector_success                           # a collector that fails drops all its metrics (0.21.0); alert PostgresExporterCollectorFailing
pg_settings_max_connections                           # denominator of the connection saturation alert
pg_stat_database_numbackends                          # numerator of the connection saturation alert
pg_stat_database_deadlocks                            # alert PostgresDeadlocks
pg_stat_database_temp_bytes                           # alert PostgresTempFilesSustained
pg_stat_database_blks_hit                             # dashboard-only: cache hit ratio is workload dependent, informational
pg_stat_database_blks_read                            # dashboard-only: with blks_hit
pg_database_size_bytes                                # dashboard-only: size and growth panel; the disk alert belongs to the volume metrics
pg_database_wraparound_age_datfrozenxid_seconds       # alerts PostgresXidWraparoundApproaching and PostgresXidWraparoundCritical (UNVERIFIED name)
pg_long_running_transactions_oldest_timestamp_seconds # alert PostgresLongRunningTransaction: age in seconds of the oldest transaction (UNVERIFIED name)
pg_stat_user_tables_n_dead_tup                        # alert PostgresDeadTuplesHigh; present once a user table exists (the scrape test creates one)
pg_stat_user_tables_n_live_tup                        # alert PostgresDeadTuplesHigh: denominator
pg_replication_lag_seconds                            # alert PostgresReplicationLagHigh; 0 on a primary, so it is always present
pg_stat_checkpointer_num_timed_total                  # alert PostgresCheckpointsRequested: scheduled checkpoints (UNVERIFIED name; PG 17+ view)
pg_stat_checkpointer_num_requested_total              # alert PostgresCheckpointsRequested: forced checkpoints (UNVERIFIED name)
pg_wal_size_bytes                                     # alert PostgresWalDirectoryLarge; needs pg_monitor (pg_ls_waldir), so revoking it removes the series
pg_wal_segments                                       # dashboard-only: with pg_wal_size_bytes; also what disappears when pg_monitor is revoked
pg_locks_count                                        # dashboard-only: lock mix by mode
pg_stat_activity_count                                # dashboard-only: sessions by state
pg_stat_statements_calls_total                        # dashboard-only: top queries by calls (query text is not exported)
pg_stat_statements_seconds_total                      # dashboard-only: top queries by total time
